package com.sg.trade.asia.domain.service;

import com.sg.trade.asia.domain.model.IngestionStatus;
import com.sg.trade.asia.domain.model.StatusUpdateRequest;
import com.sg.trade.asia.domain.port.IngestionStatusPort;
import com.sg.trade.asia.domain.port.StatusUpdatePort;

import java.time.Clock;
import java.time.LocalDate;
import java.util.List;

public class ReplayFollowUpService {

    private final IngestionStatusPort ingestionStatusPort;
    private final StatusUpdatePort statusUpdatePort;

    private final Integer projectId;
    private final String path;
    private final String optionName;
    private final Clock clock;

    public ReplayFollowUpService(
            IngestionStatusPort ingestionStatusPort,
            StatusUpdatePort statusUpdatePort,
            Integer projectId,
            String path,
            String optionName,
            Clock clock
    ) {
        this.ingestionStatusPort = ingestionStatusPort;
        this.statusUpdatePort = statusUpdatePort;
        this.projectId = projectId;
        this.path = path;
        this.optionName = optionName;
        this.clock = clock;
    }

    public void execute(
            Long ingestionId,
            String sessionId,
            LocalDate processingDate
    ) {

        String status = "KO";
        String reason;

        try {
            List<IngestionStatus> records =
                    ingestionStatusPort.getStatus(
                            ingestionId,
                            processingDate
                    );

            IngestionStatus matchingRecord = records == null
                    ? null
                    : records.stream()
                        .filter(record ->
                                sessionId.equals(record.sessionId()))
                        .findFirst()
                        .orElse(null);

            if (matchingRecord == null) {
                reason = "No matching INGaaS status record found for sessionId: "
                        + sessionId;
            } else if ("SUCCESS".equalsIgnoreCase(
                    matchingRecord.status())) {

                status = "OK";
                reason = null;

            } else {
                reason = matchingRecord.message() == null
                        || matchingRecord.message().isBlank()
                        ? "INGaaS returned status: "
                            + matchingRecord.status()
                        : matchingRecord.message();
            }

        } catch (Exception exception) {
            reason = "Follow-up status check failed: "
                    + exception.getMessage();
        }

        StatusUpdateRequest request = new StatusUpdateRequest(
                projectId,
                path,
                LocalDate.now(clock).toString(),
                optionName,
                status,
                reason
        );

        statusUpdatePort.updateStatus(request);
    }
}

package com.sg.trade.asia.infrastructure.scheduler;

import com.sg.trade.asia.domain.port.IngestionReplayPort;
import com.sg.trade.asia.domain.port.ReplaySchedulerPort;
import com.sg.trade.asia.domain.port.FollowUpSchedulerPort;

import java.time.Duration;
import java.time.LocalDate;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

import org.springframework.stereotype.Component;

@Component
public class ReplaySchedulerAdapter implements ReplaySchedulerPort {

    private final ScheduledExecutorService executor;
    private final IngestionReplayPort replayPort;
    private final ReplayScheduleRegistry registry;
    private final FollowUpSchedulerPort followUpSchedulerPort;

    public ReplaySchedulerAdapter(
            ScheduledExecutorService executor,
            IngestionReplayPort replayPort,
            ReplayScheduleRegistry registry,
            FollowUpSchedulerPort followUpSchedulerPort
    ) {
        this.executor = executor;
        this.replayPort = replayPort;
        this.registry = registry;
        this.followUpSchedulerPort = followUpSchedulerPort;
    }

    @Override
    public boolean scheduleReplay(
            Long ingestionId,
            String sessionId,
            LocalDate date,
            Duration delay
    ) {

        if (sessionId == null || sessionId.isBlank()) {
            return false;
        }

        if (!registry.register(ingestionId, sessionId)) {
            return false;
        }

        try {
            executor.schedule(() -> {
                try {
                    // Step 1: Execute the existing replay API.
                    replayPort.replay(ingestionId, sessionId);

                    // Step 2: Schedule follow-up only if replay succeeds.
                    // The 45-minute timer starts here.
                    followUpSchedulerPort.scheduleFollowUp(
                            ingestionId,
                            sessionId,
                            date
                    );

                } finally {
                    // Preserve your existing duplicate-scheduling cleanup.
                    registry.remove(ingestionId, sessionId);
                }

            }, delay.toMillis(), TimeUnit.MILLISECONDS);

            return true;

        } catch (RuntimeException exception) {
            registry.remove(ingestionId, sessionId);
            throw exception;
        }
    }
}
