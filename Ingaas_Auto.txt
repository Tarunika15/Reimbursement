@Component
public class IngaasSessionRegistry implements IngestionSessionPort {

    private final Map<SessionKey, String> sessions =
            new ConcurrentHashMap<>();

    @Override
    public void register(
            Long ingestionId,
            LocalDate ingestionDate,
            String sessionId) {

        sessions.put(
                new SessionKey(
                        ingestionId,
                        ingestionDate),
                sessionId);
    }

    @Override
    public Optional<String> getSessionId(
            Long ingestionId,
            LocalDate ingestionDate) {

        return Optional.ofNullable(
                sessions.get(
                        new SessionKey(
                                ingestionId,
                                ingestionDate)));
    }

    private record SessionKey(
            Long ingestionId,
            LocalDate ingestionDate) {
    }
}

import java.time.Instant;

@Component
@RequiredArgsConstructor
public class ReplaySchedulerAdapter
        implements ReplaySchedulerPort {

    private final TaskScheduler taskScheduler;

    private final IngestionReplayPort ingestionReplayPort;

    @Override
    public void scheduleReplay(
            Long ingestionId,
            String sessionId,
            Duration delay) {

        Instant executionTime =
                Instant.now().plus(delay);

        taskScheduler.schedule(
                () -> ingestionReplayPort.replay(
                        ingestionId,
                        sessionId),
                executionTime);
    }
}

package your.package.domain.service;

import your.package.domain.model.IngestionDefinition;
import your.package.domain.model.IngestionResult;
import your.package.domain.model.IngestionStatus;
import your.package.domain.model.StatusCheckResult;
import your.package.domain.port.IngestionSessionPort;
import your.package.domain.port.IngestionStatusPort;
import your.package.domain.port.ReplaySchedulePort;
import your.package.domain.port.ReplaySchedulerPort;

import java.time.Duration;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.Optional;

public class DefaultIngestionStatusService
        implements IngestionStatusService {

    private static final Duration REPLAY_DELAY =
            Duration.ofMinutes(10);

    private final IngestionStatusPort ingestionStatusPort;

    private final IngestionSessionPort ingestionSessionPort;

    private final ReplaySchedulerPort replaySchedulerPort;

    private final ReplaySchedulePort replaySchedulePort;

    private final List<IngestionDefinition> ingestionDefinitions;

    public DefaultIngestionStatusService(
            IngestionStatusPort ingestionStatusPort,
            IngestionSessionPort ingestionSessionPort,
            ReplaySchedulerPort replaySchedulerPort,
            ReplaySchedulePort replaySchedulePort,
            List<IngestionDefinition> ingestionDefinitions) {

        this.ingestionStatusPort =
                ingestionStatusPort;

        this.ingestionSessionPort =
                ingestionSessionPort;

        this.replaySchedulerPort =
                replaySchedulerPort;

        this.replaySchedulePort =
                replaySchedulePort;

        this.ingestionDefinitions =
                ingestionDefinitions;
    }

    @Override
    public StatusCheckResult checkStatus(
            LocalDate date) {

        List<IngestionResult> results =
                new ArrayList<>();

        for (IngestionDefinition ingestion :
                ingestionDefinitions) {

            results.add(
                    checkIngestion(
                            ingestion,
                            date));
        }

        boolean allSuccessful =
                results.stream()
                        .allMatch(result ->
                                "SUCCESS".equalsIgnoreCase(
                                        result.status()));

        String overallStatus =
                allSuccessful
                        ? "SUCCESS"
                        : "FAILED";

        String message =
                allSuccessful
                        ? "All ingestions are successfully loaded"
                        : "One or more ingestions failed";

        return new StatusCheckResult(
                overallStatus,
                message,
                results);
    }

    private IngestionResult checkIngestion(
            IngestionDefinition ingestion,
            LocalDate date) {

        List<IngestionStatus> statuses =
                ingestionStatusPort.getStatus(
                        ingestion.ingestionId(),
                        date);

        if (statuses == null ||
                statuses.isEmpty()) {

            return new IngestionResult(
                    ingestion.ingestionId(),
                    ingestion.datasetName(),
                    "UNKNOWN",
                    List.of(
                            "No status information was returned by INGaaS"));
        }

        List<IngestionStatus> failedStatuses =
                statuses.stream()
                        .filter(status ->
                                !status.isSuccessful())
                        .toList();

        if (failedStatuses.isEmpty()) {

            return new IngestionResult(
                    ingestion.ingestionId(),
                    ingestion.datasetName(),
                    "SUCCESS",
                    List.of());
        }

        scheduleReplayIfRequired(
                ingestion.ingestionId(),
                date);

        List<String> messages =
                failedStatuses.stream()
                        .map(this::getMessage)
                        .toList();

        String status =
                failedStatuses.get(0).status();

        return new IngestionResult(
                ingestion.ingestionId(),
                ingestion.datasetName(),
                status,
                messages);
    }

    private void scheduleReplayIfRequired(
            Long ingestionId,
            LocalDate date) {

        Optional<String> sessionId =
                ingestionSessionPort.getSessionId(
                        ingestionId,
                        date);

        if (sessionId.isEmpty()) {
            return;
        }

        boolean newlyScheduled =
                replaySchedulePort.markIfNotScheduled(
                        ingestionId,
                        date);

        if (!newlyScheduled) {
            return;
        }

        replaySchedulerPort.scheduleReplay(
                ingestionId,
                sessionId.get(),
                REPLAY_DELAY);
    }

    private String getMessage(
            IngestionStatus status) {

        if (status.message() == null ||
                status.message().isBlank()) {

            return "Ingestion failed without an error message";
        }

        return status.message();
    }
}
