package com.yourcompany.tfds.infrastructure.client;

import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;

@FeignClient(
        name = "ingaasReplayFeignClient",
        url = "${ingaas.url}"
)
public interface IngaasReplayFeignClient {

    @PostMapping("/ingestion/pipelines/{ingestionId}/replay")
    void replay(
            @PathVariable("ingestionId") Long ingestionId,
            @RequestParam("sessionId") String sessionId);
}


package com.yourcompany.tfds.domain.port;

public interface IngestionReplayPort {

    void replay(Long ingestionId, String sessionId);
}

package com.yourcompany.tfds.infrastructure.adapter;

import com.yourcompany.tfds.domain.port.IngestionReplayPort;
import com.yourcompany.tfds.infrastructure.client.IngaasReplayFeignClient;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

@Component
@RequiredArgsConstructor
public class IngaasReplayAdapter implements IngestionReplayPort {

    private final IngaasReplayFeignClient feignClient;

    @Override
    public void replay(Long ingestionId, String sessionId) {
        feignClient.replay(ingestionId, sessionId);
    }
}

package com.yourcompany.tfds.domain.port;

import java.time.Duration;

public interface ReplaySchedulerPort {

    void scheduleReplay(
            Long ingestionId,
            String sessionId,
            Duration delay);
}

package com.yourcompany.tfds.infrastructure.scheduler;

import com.yourcompany.tfds.domain.port.IngestionReplayPort;
import com.yourcompany.tfds.domain.port.ReplaySchedulerPort;
import lombok.RequiredArgsConstructor;
import org.springframework.scheduling.TaskScheduler;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.time.Instant;

@Component
@RequiredArgsConstructor
public class ReplaySchedulerAdapter implements ReplaySchedulerPort {

    private final TaskScheduler taskScheduler;
    private final IngestionReplayPort ingestionReplayPort;

    @Override
    public void scheduleReplay(
            Long ingestionId,
            String sessionId,
            Duration delay) {

        Instant executionTime =
                Instant.now().plus(delay);

        taskScheduler.schedule(
                () -> ingestionReplayPort.replay(
                        ingestionId,
                        sessionId),
                executionTime);
    }
}


package com.yourcompany.tfds.bootstrap.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.scheduling.concurrent.ThreadPoolTaskScheduler;

@Configuration
public class ReplaySchedulerConfiguration {

    @Bean
    public ThreadPoolTaskScheduler replayTaskScheduler() {

        ThreadPoolTaskScheduler scheduler =
                new ThreadPoolTaskScheduler();

        scheduler.setPoolSize(5);
        scheduler.setThreadNamePrefix("ingaas-replay-");
        scheduler.initialize();

        return scheduler;
    }
}


@Configuration
public class IngestionStatusConfiguration {

    @Bean
    public IngestionStatusService ingestionStatusService(
            IngestionStatusPort ingestionStatusPort,
            ReplaySchedulerPort replaySchedulerPort,
            IngestionSessionPort ingestionSessionPort,
            List<IngestionDefinition> ingestionDefinitions) {

        return new DefaultIngestionStatusService(
                ingestionStatusPort,
                replaySchedulerPort,
                ingestionSessionPort,
                ingestionDefinitions);
    }
}
